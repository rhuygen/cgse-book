% Callout boxes for standalone background explanations that aren't CGSE-specific
% (e.g. "what is a Python namespace package") -- authored in the Markdown source
% as a Pandoc fenced div, e.g.:
%
%   ::: {.concept}
%   **Python background: namespace packages.** ...
%   :::
%
% The div-to-environment conversion itself happens in latex/div-environments.lua
% (a Pandoc Lua filter, since Pandoc does not do this automatically -- verified
% empirically against pandoc 3.10.1, contrary to a common assumption). This file
% only defines what the resulting \begin{concept}...\end{concept} looks like.
%
% Reuses mdframed, which eisvogel.latex already loads for its own blockquote
% styling (see eisvogel.latex's "% blockquote" section) -- no new package
% dependency introduced here. Loaded explicitly below too: pandoc inserts
% --include-in-header content (this file) earlier in the preamble than
% eisvogel's own \usepackage{mdframed} line, so \newmdenv isn't defined yet
% if we rely on the template to load it first (verified: build fails with
% "Undefined control sequence \newmdenv" without this line).
\usepackage{mdframed}
\definecolor{concept-border}{RGB}{91,155,213}
\definecolor{concept-bg}{RGB}{235,243,251}
\newmdenv[
  leftline=true,
  rightline=false,
  topline=false,
  bottomline=false,
  linewidth=3pt,
  linecolor=concept-border,
  backgroundcolor=concept-bg,
  innerleftmargin=10pt,
  innerrightmargin=10pt,
  innertopmargin=8pt,
  innerbottommargin=8pt,
  skipabove=\parskip,
  skipbelow=\parskip
]{concept}
